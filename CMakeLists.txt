cmake_minimum_required(VERSION 3.24)
project(MechanicsLab VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
endif()
add_library(physics src/physics.cpp)
target_include_directories(physics PUBLIC src)
add_executable(physics_checks tests/physics_checks.cpp)
target_link_libraries(physics_checks PRIVATE physics)
enable_testing()
add_test(NAME physics_checks COMMAND physics_checks)
add_executable(experiments src/experiments.cpp)
target_link_libraries(experiments PRIVATE physics)
option(BUILD_GUI "Build SFML application" ON)
if(BUILD_GUI)
  if(EXISTS "${CMAKE_SOURCE_DIR}/vendor/SFML-3.0.2/lib/cmake/SFML")
    list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_SOURCE_DIR}/vendor/SFML-3.0.2")
  endif()
  find_package(SFML 3 COMPONENTS Graphics Window System QUIET)
  if(NOT SFML_FOUND)
    include(FetchContent)
    set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SFML URL https://github.com/SFML/SFML/archive/refs/tags/3.0.2.zip)
    FetchContent_MakeAvailable(SFML)
  endif()
  add_executable(MechanicsLab WIN32 src/main.cpp)
  target_link_libraries(MechanicsLab PRIVATE physics SFML::Graphics SFML::Window SFML::System)
  if(MSVC)
    target_link_options(MechanicsLab PRIVATE /ENTRY:mainCRTStartup)
  endif()
  add_custom_command(TARGET MechanicsLab POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:MechanicsLab>/assets")
  if(WIN32)
    add_custom_command(TARGET MechanicsLab POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:MechanicsLab> $<TARGET_FILE_DIR:MechanicsLab> COMMAND_EXPAND_LISTS)
  endif()
endif()
